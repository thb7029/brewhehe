function draw() { ctx.clearRect(0, 0, canvas.width, canvas.height); rings.forEach(r => { ctx.beginPath(); // 描画：開始角と終了角を計算 ctx.arc(cx, cy, r.r, r.gap + r.angle + r.gapSize, r.gap + r.angle, false); const isTarget = isDebug && (r.index === currentTargetIndex); let lightness = r.l; // ターゲット中、または壊れそうな時に少し光らせる演出（ヒット数に応じて輝度アップ） if (isTarget) lightness = 80; if (r.hitCount > 15) lightness += (r.hitCount - 15) * 2; ctx.strokeStyle = `hsl(${r.h}, ${r.s}%, ${lightness}%)`; ctx.lineWidth = isTarget ? 7 : 4; ctx.lineCap = "round"; ctx.stroke(); }); // ボールの描画 ctx.beginPath(); ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2); ctx.fillStyle = "white"; ctx.shadowBlur = 10; // ほんのり光らせる ctx.shadowColor = "white"; ctx.fill(); ctx.shadowBlur = 0; // 他の描画に影響しないよう戻す }
